# Зависимости дневника — работают и на Windows, и на Linux.
#
# ВАЖНО про шифрованную базу.
#   Драйвер шифрованной SQLite — пакет sqlcipher3 версии 0.6.x.
#   Он публикует готовые wheel-файлы на PyPI для:
#       Windows : win_amd64 / win32 / win_arm64
#       Linux   : manylinux (x86_64/aarch64/i686) и musllinux
#   Устаревший пакет sqlcipher3-binary колёс для Windows НЕ имеет —
#   именно из-за него установка на Windows падала. Не используйте его.
#   Формат зашифрованной базы у обоих пакетов одинаковый (SQLCipher 4),
#   поэтому существующие записи читаются без миграции.
sqlcipher3==0.6.2

# Полноэкранный интерфейс (модуль curses) есть в стандартной библиотеке
# Linux/macOS, но не в CPython для Windows — там нужен отдельный пакет.
windows-curses>=2.3.1 ; sys_platform == "win32"

# Синхронизация с Google Drive. Необязательно: без этих пакетов дневник
# полностью работает, просто команда синхронизации будет недоступна.
google-api-python-client>=2.100
google-auth-oauthlib>=1.1
google-auth-httplib2>=0.2
